cmake_minimum_required(VERSION 3.22)
project(CroccoSpectrum VERSION 0.1.6 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_AUTOMOC ON)
include(CTest)
find_package(Qt6 6.4 REQUIRED COMPONENTS Core Gui Widgets)
find_package(PkgConfig REQUIRED)
pkg_check_modules(FFTW REQUIRED IMPORTED_TARGET fftw3)
find_package(Threads REQUIRED)

option(RF_ENABLE_ASAN "Enable address and undefined-behavior sanitizers" OFF)
option(RF_ENABLE_TSAN "Enable the thread sanitizer (separate from ASAN)" OFF)
if(RF_ENABLE_ASAN AND RF_ENABLE_TSAN)
    message(FATAL_ERROR "Use separate builds for ASAN and TSAN")
endif()
if(RF_ENABLE_ASAN)
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined)
elseif(RF_ENABLE_TSAN)
    add_compile_options(-fsanitize=thread -fno-omit-frame-pointer)
    add_link_options(-fsanitize=thread)
endif()

add_library(rf_core
    src/recording/Recording.cpp src/recording/Metadata.cpp
    src/dsp/Spectrum.cpp src/app/Analysis.cpp src/app/Session.cpp)
target_include_directories(rf_core PUBLIC src)
target_link_libraries(rf_core PUBLIC Qt6::Core PkgConfig::FFTW Threads::Threads)
target_compile_definitions(rf_core PUBLIC RF_VERSION="${PROJECT_VERSION}" _FILE_OFFSET_BITS=64)
target_compile_options(rf_core PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)

add_library(rf_ui src/app/AnalysisController.cpp src/ui/Plots.cpp
    src/ui/ImportDialog.cpp src/ui/MainWindow.cpp)
target_link_libraries(rf_ui PUBLIC rf_core Qt6::Widgets)
target_compile_options(rf_ui PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
# Embed the original artwork so installed/portable binaries need no asset paths.
qt_add_resources(rf_ui branding
    PREFIX "/branding" BASE "assets"
    FILES assets/CroccoSpectrumIcon.png assets/CroccoSpectrumLogo.png)
add_executable(croccospectrum src/main.cpp)
target_link_libraries(croccospectrum PRIVATE rf_ui)
add_executable(croccospectrum-cli src/cli.cpp)
target_link_libraries(croccospectrum-cli PRIVATE rf_core)

if(BUILD_TESTING)
    find_package(Qt6 REQUIRED COMPONENTS Test)
    add_executable(rf-core-tests tests/CoreTests.cpp)
    target_link_libraries(rf-core-tests PRIVATE rf_core Qt6::Test)
    add_test(NAME core COMMAND rf-core-tests)
    add_executable(rf-ui-tests tests/UiTests.cpp)
    target_link_libraries(rf-ui-tests PRIVATE rf_ui Qt6::Test)
    add_test(NAME ui COMMAND rf-ui-tests)
    set_tests_properties(ui PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
    add_executable(rf-soak-test tests/Soak.cpp)
    target_link_libraries(rf-soak-test PRIVATE rf_ui Qt6::Test)
    find_package(Python3 COMPONENTS Interpreter)
    if(Python3_FOUND)
        add_test(NAME parser-stress COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/reference/fuzz_metadata.py $<TARGET_FILE:croccospectrum-cli>)
        add_test(NAME reference COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/reference/check_scipy.py $<TARGET_FILE:croccospectrum-cli>)
        set_tests_properties(reference PROPERTIES SKIP_RETURN_CODE 77)
    endif()
endif()

install(TARGETS croccospectrum croccospectrum-cli RUNTIME DESTINATION bin)
install(FILES packaging/croccospectrum.desktop DESTINATION share/applications)
install(FILES assets/CroccoSpectrumIcon.png DESTINATION share/pixmaps RENAME croccospectrum.png)
install(FILES README.md CHANGELOG.md FEATURE_STATUS.md LICENSE DESTINATION share/doc/croccospectrum)
